/* Making Guns of Tien's Ammo Maker parts. */

module Base {

	/* The dum-dum way: file the jacket off a full metal jacket bullet's nose so the lead core shows. */
	craftRecipe GunsOfTien_MakeSoftPointBullet
	{
		timedAction = Making,
		Time = 10,
		Tags = InHandCraft,
		NeedToBeLearn = False,
		category = AmmoMaker,
		ToolTip = Tooltip_Recipe_GunsOfTien_MakeSoftPointBullet,
		inputs
		{
			item 1 [ammomaker.ammomaker_BulletRifleS;ammomaker.ammomaker_BulletRifleM] mappers[bulletType] mode:destroy,
			item 1 [Base.SmallFileSet] mode:keep,
		}
		outputs
		{
			item 1 mapper:bulletType,
		}
		itemMapper bulletType
		{
			GunsOfTien.BulletRifleSSP = ammomaker.ammomaker_BulletRifleS,
			GunsOfTien.BulletRifleMSP = ammomaker.ammomaker_BulletRifleM,
		}
	}

	/* Molten metal poured through a strainer into water forms small round shot (the shot-pan method). Same metal,
	   fuel and output as Ammo Maker's Make / Produce Balls 00, a strainer and water instead of a mold. */
	craftRecipe GunsOfTien_MakeBirdshot
	{
		timedAction = Making,
		Time = 200,
		Tags = PrimitiveFurnace;CanBeDoneFromFloor,
		NeedToBeLearn = False,
		category = AmmoMakerParts,
		xpAward = Blacksmith:10,
		ToolTip = Tooltip_Recipe_GunsOfTien_MakeBirdshot,
		inputs
		{
			item 1 tags[base:smeltableironsmall] mode:destroy flags[ItemCount;AllowDestroyedItem],
			item 1 [Base.CeramicCrucible;Base.CeramicCrucibleSmall] mode:keep flags[IsEmpty],
			item 1 tags[base:crudetongs;base:tongs] mode:keep,
			item 1 tags[base:charcoal],
			item 1 [Base.Strainer] mode:keep flags[MayDegradeLight],
			item 1 [*] mode:keep flags[ItemIsFluid;HandcraftOnly],
			-fluid 0.5 categories[Water] mode:mixture,
		}
		outputs
		{
			item 3 GunsOfTien.Birdshot,
		}
	}

	craftRecipe GunsOfTien_ProduceBirdshot
	{
		timedAction = Making,
		Time = 150,
		Tags = Furnace;CanBeDoneFromFloor,
		NeedToBeLearn = False,
		category = AmmoMakerParts,
		xpAward = Blacksmith:10,
		ToolTip = Tooltip_Recipe_GunsOfTien_ProduceBirdshot,
		inputs
		{
			item 1 [Base.CeramicCrucible_Iron;Base.CeramicCrucibleSmall_Iron],
			item 1 tags[base:crudetongs;base:tongs] mode:keep,
			item 1 tags[base:charcoal],
			item 1 [Base.Strainer] mode:keep flags[MayDegradeLight],
			item 1 [*] mode:keep flags[ItemIsFluid;HandcraftOnly],
			-fluid 0.5 categories[Water] mode:mixture,
		}
		outputs
		{
			item 3 GunsOfTien.Birdshot,
		}
	}
}
